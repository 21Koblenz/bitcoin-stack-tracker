#include <tunables/global>

profile bitcoin_stack_tracker_dashboard flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  file,
  capability chown,
  capability dac_override,
  capability fowner,
  # run.sh is root while Tor runs as uid 100. Linux requires CAP_KILL for
  # cross-UID TERM/KILL even when AppArmor signal mediation also permits it.
  capability kill,
  capability net_admin,
  capability setgid,
  capability setuid,
  signal (send) set=(kill,term,int,hup,cont,exists),
  signal (receive) set=(kill,term,int,quit,hup,usr1,usr2,cont,exists),

  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /usr/sbin/** ix,
  /sbin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /usr/lib/bashio/** ix,
  /tmp/** rwk,
  /data/tor/** rwk,
  /dev/tty rw,
  /proc/net/** r,

  # The gateway does not need Home Assistant configuration, Docker control,
  # or arbitrary process memory. Keep those out of the root/Tor profile too.
  deny /config/** rwklx,
  deny /homeassistant/** rwklx,
  deny /var/run/docker.sock rwklx,
  deny /proc/*/mem rwklx,
  deny /proc/kcore rwklx,

  # Tor is the only process allowed by nftables to initiate public traffic.
  network inet stream,
  network inet dgram,
  network inet6 stream,
  network inet6 dgram,
  network unix stream,

  /app/network_agent.py cx -> network_agent,

  profile network_agent flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    signal (receive) peer=*_bitcoin_stack_tracker_dashboard,
    signal (receive, send) set=(exists),
    network inet stream,
    network inet6 stream,

    /usr/bin/python3* rix,
    /usr/lib/** mr,
    /lib/** mr,
    /app/** r,
    /run/bitcoin-stack-network-status.json r,
    /dev/null rw,
    /dev/urandom r,

    # The health agent has no reason to read HA config, portfolio data or Tor state.
    deny /config/** rwklx,
    deny /homeassistant/** rwklx,
    deny /data/** rwklx,
    deny /var/run/docker.sock rwklx,
    deny /bin/** x,
    deny /usr/bin/sh x,
    deny /usr/bin/bash x,
    deny /usr/sbin/nft x,
    deny /sbin/nft x,
  }
}
